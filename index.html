Pengaturan Integrasi Google Sheets

Hubungkan web app ini dengan spreadsheet Google Anda melalui Google Apps Script Web App.

Cara Menghubungkan Google Sheets:

  1. Buka Google Sheets baru Anda. Buat 2 sheet bernama Produk dan Penjualan.
  2. Di sheet Produk buat kolom header: Kode | Nama | Kategori | Harga | Stok | Gambar.
  3. Di sheet Penjualan buat kolom header: ID Transaksi | Tanggal | Items | Total | Bayar | Kembalian.
  4. Klik menu Extensions / Ekstensi > Apps Script.
  5. Copas kode Google Apps Script di bawah ini lalu klik Deploy > New Deployment > Web App. Set "Who has access" ke Anyone.
  6. Salin Web App URL dan tempelkan ke kolom input di atas.
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Produk");
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var result = [];
  
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j].toString().toLowerCase()] = data[i][j];
    }
    result.push(obj);
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Penjualan");
  
  sheet.appendRow([
    data.id,
    data.tanggal,
    JSON.stringify(data.items),
    data.total,
    data.bayar,
    data.kembalian
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
